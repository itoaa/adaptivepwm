# Cppcheck Suppressions for AdaptivePWM
# These are intentional design decisions or false positives

# ============================================
# Embedded System Suppressions
# ============================================

# Hardware register access - intentional direct memory access
noPointerArithmetic:*hal_*.c
noPointerArithmetic:src/*_hal_*.c

# Volatile pointer access - required for hardware registers
noConstParameter:*hal_*.c
noConstParameter:src/hal_*.c

# ============================================
# Security Design Suppressions
# ============================================

# PBKDF2 intentionally uses configurable iteration count (SEC-027)
# This is verified in tests/test_cli_auth.c
variableScope:src/cli_auth.c:pbkdf2_sha256

# Memory allocation suppression for embedded flash logger
# Flash logger intentionally writes to fixed memory addresses
memleak:src/flash_logger.c

# Null pointer check - false positive in hal_pwm.c
# Bounds are checked before access
nullPointer:src/hal_pwm.c

# Unsigned less than zero check - intentional for clarity
unsignedLessThanZero:src/pid_controller.c

# ============================================
# False Positive Suppressions
# ============================================

# Sizeof check - intentional use of sizeof on pointer
sizeofsizeof:src/cli_commands.c

# Copying data to char array - intentional for CLI parsing
strcpy:src/cli_commands.c

# ============================================
# Known Safe Patterns
# ============================================

# Hardware RNG may not be available in all test environments
# Fallback is documented and secured (SEC-033)
uninitvar:src/cli_auth.c:secure_random

# Function pointer for callbacks - intentional design
funcArg:src/freertos_tasks.c
